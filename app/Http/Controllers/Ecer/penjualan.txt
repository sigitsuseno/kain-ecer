<?php

namespace App\Http\Controllers\Ecer;

use App\Http\Controllers\Controller;
use App\Models\Pelanggan;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class TransaksiPenjualanEceranController extends Controller
{
    public function list()
    {
        return inertia('Ecer/Penjualan/List');
    }

    public function index(Request $request)
    {

        $penjualan = Penjualan::with([
            'items.produk_varian.produk',
            'items.produk_varian.stokFisiks',
            'pelanggan',
            'marketing'
        ])
            // ->where('user_id', auth()->id())
            ->where('status_penjualan', 'DRAFT')
            ->first();

        return inertia('Ecer/Penjualan/Kasir', [
            'draft' => $penjualan,
            'pelanggan' => Pelanggan::orderBy('nama')->get(),
            'marketing' => Supplier::where('status', 'marketing')->get(),
            'produk' => ProdukVarian::with([
                'produk',
                'stokFisiks' => fn($q) => $q->where('total_panjang', '>', 0),
                'detailIsiPaket.satuan',
            ])->get(),
        ]);
    }

    public function storeDraft(Request $request)
    {
        // dd($request);
        $data = $request->validate([
            'tanggal_transaksi' => 'required|date',
            'pelanggan' => 'required|string',
            'marketing' => 'nullable|string',
        ]);

        return DB::transaction(function () use ($data) {

            $slugPelanggan = Str::slug($data['pelanggan']);

            $pelanggan = Pelanggan::firstOrCreate(
                ['slug' => $slugPelanggan],
                ['nama' => $data['pelanggan']]
            );


            $marketing = null;

            if (!empty($data['marketing'])) {
                $slugMarketing = Str::slug($data['marketing']);

                $marketing = Supplier::firstOrCreate(
                    ['slug' => $slugMarketing],
                    [
                        'nama' => $data['marketing'],
                        'status' => 'marketing'
                    ]
                );
            }



            $penjualan = Penjualan::updateOrCreate(
                [
                    'user_id' => Auth::id(),
                    'status_penjualan' => 'DRAFT',
                ],
                [
                    'nomer_nota' => $this->generateNota(),
                    'tanggal_transaksi' => $data['tanggal_transaksi'],
                    'pelanggan_id' => $pelanggan->id,
                    'marketing_id' => $marketing?->id,
                    'total_harga' => 0,
                    'diskon' => 0,
                    'pajak' => 0,
                    'biaya_kirim' => 0,
                    'total_akhir' => 0,
                    'metode_bayar' => null,
                ]
            );

            return redirect()->back()->with('draft_id', $penjualan->id);
        });
    }

    protected function generateNota(): string
    {
        $prefix = 'INV-' . now()->format('Ym');

        $last = Penjualan::where('nomer_nota', 'like', "$prefix%")
            ->orderByDesc('id')
            ->value('nomer_nota');

        $number = $last
            ? (int) substr($last, -4) + 1
            : 1;

        return $prefix . '-' . str_pad($number, 4, '0', STR_PAD_LEFT);
    }
}

<?php

namespace App\Http\Controllers\Ecer;

use App\Http\Controllers\Controller;
use App\Models\Penjualan;
use App\Models\Pembayaran;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class PembayaranEcerController extends Controller
{
    public function bayar(Request $request, Penjualan $penjualan)
    {
        $data = $request->validate([
            'tanggal_bayar' => 'required|date',
            'metode_bayar' => 'required|string',
            'total_bayar' => 'required|numeric|min:0',
            'details' => 'array',
        ]);

        DB::transaction(function () use ($penjualan, $data) {

            $pembayaran = Pembayaran::create([
                'penjualan_id' => $penjualan->id,
                'tanggal_bayar' => $data['tanggal_bayar'],
                'total_bayar' => $data['total_bayar'],
                'jenis_pembayaran' => $data['metode_bayar'],
            ]);

            foreach ($data['details'] ?? [] as $detail) {
                $pembayaran->detail_pembayarans()->create($detail);
            }

            $penjualan->update([
                'status_penjualan' => 'PAID',
                'total_akhir' => $data['total_bayar'],
            ]);
        });

        return redirect()->route('ecer.penjualan.print', $penjualan);
    }
}

<script setup>
import { ref, computed } from 'vue'
import { Link, router } from '@inertiajs/vue3'
import AuthLayout from '../../../Layouts/AuthLayout.vue';

defineOptions({ layout: AuthLayout })

const props = defineProps({
    draft: Object,
    pelanggan: Array,
    marketing: Array,
    produk: Array,
})

const today = new Date().toLocaleDateString('id-ID')

const step = ref(props.draft ? 3 : 1)
const selectedRoll = ref(null);
const inputHarga = ref(0);

const draftForm = ref({
    tanggal_transaksi: new Date().toISOString().slice(0, 10),
    pelanggan: '',
    marketing: '',
})

const payment = ref({
    tanggal_bayar: new Date().toISOString().slice(0, 10),
    metode: 'CASH',
    total_bayar: 0,
    bank_asal: '',
    ref: '',
})

const submitDraft = () => {
    router.post('/ecer/penjualan/transaksi/draft', draftForm.value, {
        onSuccess: () => step.value = 3
    })
}

// Di dalam script setup Anda

const hapusPenjualan = (id) => {
    if (confirm('Apakah Anda yakin ingin membatalkan transaksi ini dan memulai dari awal?')) {
        router.delete(`/ecer/penjualan/transaksi/${id}`, {
            onSuccess: () => {
                // Reset state ke awal
                step.value = 1;
                cart.value = [];
                draftForm.value = {
                    tanggal_transaksi: new Date().toISOString().slice(0, 10),
                    pelanggan: '',
                    marketing: '',
                };
            }
        });
    }
}

const cart = ref([]);

const addRoll = (varian, roll) => {
    // 1. Cari apakah Varian (Produk + Warna) ini sudah ada di keranjang?
    let itemProduk = cart.value.find(item => item.variant_id === varian.id);

    if (itemProduk) {
        // 2. Jika Varian sudah ada, cek apakah Roll ini sudah dimasukkan?
        const isRollExist = itemProduk.details.find(d => d.roll_id === roll.id);

        if (!isRollExist) {
            // Tambahkan roll ke dalam list details varian tersebut
            itemProduk.details.push({
                roll_id: roll.id,
                nomor_batch: roll.items_details?.nomor_batch || 'N/A', // contoh ambil dr StokFisik
                qty: roll.qty, // panjang asli roll
                subtotal: roll.qty * itemProduk.harga // subtotal per roll
            });
        }
    } else {
        // 3. Jika Varian belum ada, buat entry baru
        cart.value.push({
            variant_id: varian.id,
            nama: varian.produk.nama_produk,
            warna: varian.nama_varian,
            harga: varian.harga_jual || 0, // harga default
            details: [{
                roll_id: roll.id,
                nomor_batch: roll.items_details?.nomor_batch || 'N/A',
                qty: roll.qty,
                subtotal: 0
            }]
        });
    }
};

const updateSubtotal = (item) => {
    item.subtotal = item.qty * item.harga
}

const grandTotal = computed(() => {
    return cart.value.reduce((total, item) => total + (item.qty * item.harga), 0)
})

const handleProsesPembayaran = () => {
    if (cart.value.length === 0) return alert("Keranjang masih kosong")

    router.post('/ecer/penjualan/transaksi/item', {
        penjualan_id: props.draft.id,
        items: cart.value
    }, {
        onSuccess: () => step.value = 4 
    })
}



const submitPayment = () => {
    router.post(`/ecer/penjualan/${props.draft.id}/bayar`, payment.value, {
        onSuccess: () => {
            alert('Pembayaran Berhasil!');
            window.print();
        }
    });
};


const format = (n) =>
    new Intl.NumberFormat('id-ID').format(n)


</script>

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Penjualan extends Model
{
    protected $fillable = [
        'nomer_nota',
        'tanggal_transaksi',
        'pelanggan_id',
        'user_id',
        'marketing_id',
        'total_harga',
        'diskon',
        'pajak',
        'biaya_kirim',
        'total_akhir',
        'metode_bayar',
        'status_penjualan',
    ];

    public function items()
    {
        return $this->hasMany(ItemPenjualan::class, 'penjualan_id');
    }

    public function pembayaran()
    {
        return $this->hasOne(Pembayaran::class, 'penjualan_id');
    }

    public function pelanggan()
    {
        return $this->belongsTo(Pelanggan::class, 'pelanggan_id');
    }

    public function user()
    {
        return $this->belongsTo(User::class, 'user_id');
    }

    public function marketing()
    {
        return $this->belongsTo(Supplier::class, 'marketing_id');
    }
}

use Illuminate\Database\Eloquent\Model;

class DetailPenjualan extends Model
{
    protected $fillable = [
        'item_penjualan_id',
        'varian_id',
        'satuan_id',
        'qty',
        'status',
    ];

    public function produk_varian()
    {
        return $this->belongsTo(ProdukVarian::class, 'varian_id');
    }

    public function itemsPenjualan()
    {
        return $this->belongsTo(ItemPenjualan::class, 'item_penjualan_id');
    }

    public function satuan()
    {
        return $this->belongsTo(Satuan::class, 'satuan_id');
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Pembayaran extends Model
{
    protected $fillable = [
        'penjualan_id',
        'nomer_pembayaran',
        'tanggal_bayar',
        'total_bayar',
        'jenis_pembayaran',
        'keterangan',
        'is_dp',
    ];

    public function penjualan()
    {
        return $this->belongsTo(Penjualan::class, 'penjualan_id');
    }

    public function detail_pembayarans()
    {
        return $this->hasMany(PembayaranDetails::class, 'pembayaran_id');
    }
}

// model ProdukVarian 

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class ProdukVarian extends Model
{
    protected $fillable = [
        'produk_id',
        'gudang_id',
        'nama_varian',
        'tipe',
        'panjang',
        'tinggi',
        'lebar',
        'tebal',
        'sku',
        'is_paket',
        'harga_beli',
        'harga_jual',
    ];

    public function produk()
    {
        return $this->belongsTo(Produk::class, 'produk_id');
    }
    public function stokFisiks()
    {
        return $this->hasMany(StokFisik::class, 'varian_id');
    }
    public function detailIsiPaket()
    {
        return $this->hasMany(DetailPaket::class, 'induk_varian_id');
    }
    public function konversi()
    {
        return $this->hasMany(KonversiSatuan::class, 'satuan_id');
    }
    public function pembelianItem()
    {
        return $this->hasOne(PembelianItems::class, 'varian_id');
    }
}

// model StokFisik

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class StokFisik extends Model
{
    protected $table = 'stok_fisiks';

    protected $fillable = [
        'varian_id',
        'gudang_id',
        'nomor_batch',
        'tanggal_kadaluarsa',
        'total_panjang',
        'total_qty',
        'harga_jual',
        'harga_beli',
    ];

    public function varian()
    {
        return $this->belongsTo(ProdukVarian::class, 'varian_id');
    }

    public function gudang()
    {
        return $this->belongsTo(Gudang::class, 'gudang_id');
    }

    public function detailPakets()
    {
        return $this->HasMany(DetailPaket::class, 'anak_varian_id');
    }
}
