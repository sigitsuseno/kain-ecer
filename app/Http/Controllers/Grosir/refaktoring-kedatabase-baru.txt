## Model Product

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Casts\Attribute;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class Product extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'category_id',
        'supplier_id',
        'kode_barang',
        'name',
        'slug',
        'tipe',
        'seri',
        'kode_warna',
        'description',
        'lebar_kain',
        'total_lenght',
        'jumlah_roll',
        'selling_price',
        'cost_price',
        'inventory_method',
        'tracking_method',
        'abc_category',
        'min_stock',
        'max_stock',
        'safety_stock',
        'reorder_point',
        'tax_rate',
        'is_active',
        'unit_satuan',
    ];

    protected $casts = [
        'min_stock' => 'decimal:4',
        'max_stock' => 'decimal:4',
        'safety_stock' => 'decimal:4',
        'reorder_point' => 'decimal:4',
        'cost_price' => 'decimal:4',
        'selling_price' => 'decimal:4',
        'tax_rate' => 'decimal:2',
        'unit_conversion_factor' => 'decimal:4',
        'weight' => 'decimal:4',
        'dimensions' => 'array',
        'has_expiry_date' => 'boolean',
        'require_quality_check' => 'boolean',
        'is_active' => 'boolean',
    ];

    // Relationships
    public function category(): BelongsTo
    {
        return $this->belongsTo(Category::class);
    }

    public function supplier(): BelongsTo
    {
        return $this->belongsTo(Supplier::class);
    }

    public function rolls()
    {
        return $this->hasMany(JumlahRoll::class, 'product_id');
    }


    public static function createWithRules(string $name, string $kodeWarna, array $data): self
    {

        $exists = self::where('name', $name)
            ->where('kode_warna', $kodeWarna)
            ->exists();

        if ($exists) {
            throw ValidationException::withMessages([
                'kode_warna' => 'Warna ini sudah ada untuk produk tersebut.',
            ]);
        }

        $data['slug'] = Str::slug($name . '-' . $kodeWarna);
        $data['name'] = $name;
        $data['kode_warna'] = $kodeWarna;

        return self::create($data);
    }

    public function updateWithRules(string $name, string $kodeWarna, array $data): bool
    {

        $exists = self::where('name', $name)
            ->where('kode_warna', $kodeWarna)
            ->where('id', '!=', $this->id)
            ->exists();

        if ($exists) {
            throw ValidationException::withMessages([
                'kode_warna' => 'Warna ini sudah ada untuk produk tersebut.',
            ]);
        }

        $data['slug'] = Str::slug($name . '-' . $kodeWarna);
        $data['name'] = $name;
        $data['kode_warna'] = $kodeWarna;

        return $this->update($data);
    }

    // Accessors
    protected function grossMargin(): Attribute
    {
        return Attribute::make(
            get: function () {
                $selling = (float) $this->selling_price;
                $cost = (float) $this->cost_price;
                if ($selling > 0) {
                    return number_format((($selling - $cost) / $selling) * 100, 2, '.', '');
                }

                return '0.00';
            }
        );
    }

    // Business Logic Methods
    public function needsReorder(): bool
    {
        return false; // Placeholder
    }

    // Scopes
    public function scopeActive($query)
    {
        return $query->where('is_active', true);
    }

    public function scopeAbcCategory($query, $category)
    {
        return $query->where('abc_category', $category);
    }

    public function scopeLowStock($query)
    {
        return $query->whereColumn('reorder_point', '>=', 'safety_stock');
    }

    public function invoiceItems()
    {
        return $this->hasMany(PurchaseInvoiceItem::class, 'product_id');
    }

    public function inventory()
    {
        return $this->hasMany(Inventory::class, 'product_id');
    }

    public function stockCountingItems()
    {
        return $this->hasMany(StockCountingItem::class, 'product_id');
    }
    public  function saleItems()
    {
        return $this->hasMany(SaleItem::class, 'product_id');
    }
}```
