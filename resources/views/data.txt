<?php

namespace App\Http\Controllers\Master;

use App\Http\Controllers\Controller;
use App\Models\Gudang;
use App\Models\Kategori;
use App\Models\Produk;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Illuminate\Validation\Rule;
use Inertia\Inertia;

class ProductController extends Controller
{
    public function index(Request $request)
    {

        $perPage = 10;

        $products = Produk::with(['kategori', 'gudang'])
            ->when(
                $request->search,
                fn($q) =>
                $q->where('nama_produk', 'like', "%{$request->search}%")
                    ->orWhere('kode_produk', 'like', "%{$request->search}%")
            )
            ->when(
                $request->kategori,
                fn($q) =>
                $q->where('kategori_id', $request->kategori)
            )
            ->orderBy('id', 'desc')
            ->paginate($perPage)
            ->withQueryString();

        return Inertia::render('Dashboard/Master', [
            'products' => $products,
            'categories' => Kategori::where('status', 'np')->get(),
            'gudangs' => Gudang::orderBy('nama_gudang')->get(),
            'allProducts' => Produk::orderBy('nama_produk')->get(),
        ]);
    }

    public function store(Request $request)
    {

        $data = $request->validate([
            'nama_produk' => 'required|string|max:255',
            'kode_warna' => [
                'required',
                'string',
                'max:100',
                Rule::unique('produks')->where(
                    fn($q) =>
                    $q->where('nama_produk', $request->nama_produk)
                ),
            ],
            'satuan' => 'required|in:meter,yard,kg',
            'gudang_id' => 'nullable|exists:gudangs,id',
            'deskripsi' => 'nullable|string',
        ]);
        // normalisasi warna
        $namaProdukClean = trim(ucwords(strtolower($request->nama_produk)));

        $kategori = Kategori::firstOrCreate(
            ['slug' => Str::slug($namaProdukClean)],
            ['nama_kategori' => $namaProdukClean]
        );

        $data['kategori_id'] = $kategori->id;
        $data['nama_produk'] = $namaProdukClean;

        $data['kode_produk'] = 'PRD-' . strtoupper(Str::random(6));
        $data['slug'] = Str::slug($request->nama_produk);

        Produk::create($data);

        return back()->with('success', 'Produk berhasil ditambahkan');
    }

    public function update(Request $request, Produk $produk)
    {
        $data = $request->validate([
            'nama_produk' => 'required|string|max:255',
            'kode_warna' => [
                'required',
                Rule::unique('produks')
                    ->ignore($produk->id)
                    ->where(
                        fn($q) =>
                        $q->where('nama_produk', $request->nama_produk)
                    ),
            ],
            'satuan' => 'required|in:meter,yard,kg',
            'gudang_id' => 'required|exists:gudangs,id',
            'deskripsi' => 'nullable|string',
        ]);

        $namaProdukClean = trim(ucwords(strtolower($request->nama_produk)));

        $kategori = Kategori::firstOrCreate(
            ['slug' => Str::slug($namaProdukClean)],
            ['nama_kategori' => $namaProdukClean]
        );

        $data['kategori_id'] = $kategori->id;
        $data['nama_produk'] = $namaProdukClean;

        $data['slug'] = Str::slug($request->nama_produk);

        $produk->update($data);

        return back()->with('success', 'Produk berhasil diperbarui');
    }
}

<script setup>
import { useForm } from '@inertiajs/vue3'
import { ref, computed, watch } from 'vue'

const props = defineProps({
    mode: String,
    product: Object,
    categories: Array,
    gudangs: Array,
    allProducts: Array,
})

const emit = defineEmits(['close'])

const isView = computed(() => props.mode === 'view')
const isEdit = computed(() => props.mode === 'edit')

// State untuk UI dropdown
const showDropdown = ref(false)

// 1. Logika Filter Kategori (Helper Pengelompokan)
const filteredCategories = computed(() => {
    const s = form.nama_produk?.toLowerCase().trim()
    if (!s || isEdit.value) return []
    return (props.categories ?? []).filter(c =>
        c.nama_kategori?.toLowerCase().includes(s) &&
        c.nama_kategori?.toLowerCase() !== s // Hilang jika sudah diketik lengkap
    )
})

// 2. Logika Cek Duplikat (Nama + Kode Warna)
const isDuplicate = computed(() => {
    if (!form.nama_produk || !form.kode_warna || isView.value) return false

    return (props.allProducts ?? []).some(p => {
        if (isEdit.value && p.id === props.product?.id) return false
        return p.nama_produk.toLowerCase() === form.nama_produk.toLowerCase() &&
            p.kode_warna.toLowerCase() === form.kode_warna.toLowerCase()
    })
})

const isColorValid = computed(() => {
    return form.nama_produk.length > 0 &&
        form.kode_warna.length > 0 &&
        !isDuplicate.value
})

const form = useForm({
    nama_produk: '',
    kode_warna: '',
    satuan: 'yard',
    tipe: 'polos',
    kode_seri: '',
    lebar_kain: '',
    gudang_id: 1,
    deskripsi: '',
})

// 3. Fungsi pilih kategori yang diperbaiki
function selectCategory(name) {
    form.nama_produk = name
    showDropdown.value = false
}

// Watcher untuk mengisi data saat Edit atau Reset saat Create
watch(
    () => props.product,
    (p) => {
        if (p) {
            form.nama_produk = p.nama_produk
            form.kode_warna = p.kode_warna
            form.kode_seri = p.kode_seri
            form.tipe = p.tipe
            form.lebar_kain = p.lebar_kain
            form.satuan = p.satuan
            form.gudang_id = p.gudang_id
            form.deskripsi = p.deskripsi
        } else {
            form.reset()
        }
    },
    { immediate: true }
)

function submit() {
    const routeName = isEdit.value
        ? route('dashboard.master.products.update', props.product.id)
        : route('dashboard.master.products.store')

    const method = isEdit.value ? 'put' : 'post'

    form[method](routeName, {
        onSuccess: () => emit('close'),
    })
}
</script>
