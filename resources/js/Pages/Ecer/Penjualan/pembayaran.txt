disini adalah tahap pembayaran yang mendapatkan data dari tabel penjualan -> item_penjualan -> detail_penjualan.

dimana pembayaran ini ada 5 mode pembayaran yaitu CASH, TRANSFER, EDC, SPLIT, DP.
1. CASH
    - field yang di butuhkan = Tanggal, Metode, Ongkir (bila ada), Diskon (bila ada), nominal bayar, kembalian.
2. TRANSFER dan EDC
     - field yang di butuhkan = Tanggal, Metode, bank_asal, bank_tujuan, Ongkir (bila ada), Diskon (bila ada), nominal bayar, kembalian.
3. SPLIT
     - field yang di butuhkan = Tanggal, Metode, bank_asal, bank_tujuan, Ongkir (bila ada), Diskon (bila ada), nominal bayar 1, nominal bayar 2 dst, kembalian.
2. DP
     - field yang di butuhkan = Tanggal, Metode, bank_asal, bank_tujuan, Ongkir (bila ada), Diskon (bila ada), nominal bayar, Kurang Bayar.

<?php

class TransaksiPenjualanEceranController extends Controller
{
    public function list(Request $request)
    {
        $query = Penjualan::with([
            'pelanggan',
            'marketing',
            'items.produk_varian.produk',
            'items.detailPenjualan'
        ])->where('status_penjualan', 'UNPAID');

        // Tambahkan filter search sederhana jika dibutuhkan
        if ($request->search) {
            $query->where('nomer_nota', 'like', "%{$request->search}%")
                ->orWhereHas('pelanggan', function ($q) use ($request) {
                    $q->where('nama', 'like', "%{$request->search}%");
                });
        }
        return inertia('Ecer/Penjualan/List', [
            'penjualan' => $query->latest()->get(),
        ]);
    }

    public function index(Request $request)
    {

        $penjualan = Penjualan::with([
            'items.produk_varian.produk',
            'items.produk_varian.stokFisiks',
            'pelanggan',
            'marketing'
        ])
            ->where('status_penjualan', 'DRAFT')
            ->first();

        return inertia('Ecer/Penjualan/Kasir', [
            'draft' => $penjualan,
            'pelanggan' => Pelanggan::orderBy('nama')->get(),
            'marketing' => Supplier::where('status', 'marketing')->get(),
            'produk' => ProdukVarian::with([
                'produk',
                'stokFisiks' => fn($q) => $q->where('total_panjang', '>', 0),
                'detailIsiPaket.satuan',
            ])->get(),
            'company' => Perusahaan::find(1) ?? null,
        ]);
    }

    public function storeDraft(Request $request)
    {
        // dd($request);
        $data = $request->validate([
            'tanggal_transaksi' => 'required|date',
            'pelanggan' => 'required|string',
            'marketing' => 'nullable|string',
        ]);

        return DB::transaction(function () use ($data) {

            $slugPelanggan = Str::slug($data['pelanggan']);

            $pelanggan = Pelanggan::firstOrCreate(
                ['slug' => $slugPelanggan],
                ['nama' => $data['pelanggan']]
            );


            $marketing = null;

            if (!empty($data['marketing'])) {
                $slugMarketing = Str::slug($data['marketing']);

                $marketing = Supplier::firstOrCreate(
                    ['slug' => $slugMarketing],
                    [
                        'nama' => $data['marketing'],
                        'status' => 'marketing'
                    ]
                );
            }



            $penjualan = Penjualan::updateOrCreate(
                [
                    'user_id' => Auth::id(),
                    'status_penjualan' => 'DRAFT',
                ],
                [
                    'nomer_nota' => $this->generateNota(),
                    'tanggal_transaksi' => $data['tanggal_transaksi'],
                    'pelanggan_id' => $pelanggan->id,
                    'marketing_id' => $marketing?->id,
                    'total_harga' => 0,
                    'diskon' => 0,
                    'pajak' => 0,
                    'biaya_kirim' => 0,
                    'total_akhir' => 0,
                    'metode_bayar' => null,
                ]
            );

            return redirect()->back()->with('draft_id', $penjualan->id);
        });
    }

    protected function generateNota(): string
    {
        $prefix = 'INV-' . now()->format('Ym');

        $last = Penjualan::where('nomer_nota', 'like', "$prefix%")
            ->orderByDesc('id')
            ->value('nomer_nota');

        $number = $last
            ? (int) substr($last, -4) + 1
            : 1;

        return $prefix . '-' . str_pad($number, 4, '0', STR_PAD_LEFT);
    }

    public function simpanItems(Request $request)
    {

        $request->validate([
            'penjualan_id' => 'required|exists:penjualans,id',
            'items' => 'required|array|min:1',
            'items.*.variant_id' => 'required|exists:produk_varians,id',
            'items.*.satuan_id' => 'required',
            'items.*.harga' => 'required|numeric|min:0',
            'items.*.details' => 'required|array|min:1',
            'items.*.details.*.roll_id' => 'required|exists:detail_pakets,id',
            'items.*.details.*.qty' => 'required|numeric|min:0.01',
        ]);

        // dd($request);

        return DB::transaction(function () use ($request) {

            $grandTotal = 0;

            foreach ($request->items as $item) {

                $totalQtyVarian = 0;

                foreach ($item['details'] as $detail) {
                    $detailPaket = DetailPaket::findOrFail($detail['roll_id']);

                    // VALIDASI STOK: Cek apakah qty yang diinput melebihi stok fisik roll
                    if ($detail['qty'] > $detailPaket->qty) {
                        throw \Illuminate\Validation\ValidationException::withMessages([
                            'items' => "Stok Roll ID {$detail['roll_id']} tidak mencukupi. Sisa: {$detailPaket->qty}, Input: {$detail['qty']}"
                        ]);
                    }

                    $detail = DetailPenjualan::create([
                        'item_penjualan_id' => $request->penjualan_id,
                        'varian_id' => $item['variant_id'],
                        'satuan_id' => $detailPaket->satuan_id,
                        'qty' => $detail['qty'],
                        'status' => 'transisi',
                    ]);

                    $samaBanyak = $detailPaket->qty - $detail['qty'] === 0;
                    if ($samaBanyak) {
                        $detailPaket->delete();
                    } else {
                        $detailPaket->decrement('qty', $detail['qty']);
                    }

                    $totalQtyVarian += $detail['qty'];
                }


                $subtotal = $totalQtyVarian * $item['harga'];

                $item_penjualan = ItemPenjualan::create([
                    'penjualan_id' => $request->penjualan_id,
                    'produk_varian_id' => $item['variant_id'],
                    'satuan_id' => $item['satuan_id'],
                    'qty' => $totalQtyVarian,
                    'harga_satuan' => $item['harga'],
                    'subtotal' => $subtotal,
                ]);

                $grandTotal += $subtotal;
            }

            Penjualan::where('id', $request->penjualan_id)->update([
                'total_harga' => $grandTotal,
                'status_penjualan' => 'UNPAID',
            ]);

            return redirect()->route('ecer.penjualan.pembayaran.prepare', $request->penjualan_id);
        });
    }

    public function hapusPenjualan(Request $request, $id)
    {
        return DB::transaction(function () use ($id) {
            $penjualan = Penjualan::findOrFail($id);

            $penjualan->items()->delete();

            $penjualan->delete();

            return redirect()->back();
        });
    }
}

namespace App\Http\Controllers\Ecer;

use App\Http\Controllers\Controller;
use App\Models\Pelanggan;

use App\Models\ProdukVarian;
use App\Models\Supplier;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class PembayaranEcerController extends Controller
{
    public function index(Request $request)
    {
        $query = Pembayaran::with([
            'pelanggan',
            'marketing',
            'items.produk_varian.produk',
            'items.detailPenjualan'
        ])->where('status_penjualan', 'UNPAID');

        // Tambahkan filter search sederhana jika dibutuhkan
        if ($request->search) {
            $query->where('nomer_nota', 'like', "%{$request->search}%")
                ->orWhereHas('pelanggan', function ($q) use ($request) {
                    $q->where('nama', 'like', "%{$request->search}%");
                });
        }

        return inertia('Ecer/Penjualan/ListPembayaran', [
            'penjualan' => $query->latest()->get(),
        ]);
    }

    public function preparePembayaran($id)
    {
        return inertia('Ecer/Penjualan/Pembayaran', [
            'penjualan' => Penjualan::with([
                'pelanggan',
                'items.produk_varian.produk',
                'items.detailPenjualan'
            ])->findOrFail($id),

            'pelanggan' => Pelanggan::orderBy('nama')->get(),
            'marketing' => Supplier::where('status', 'marketing')->get(),
            'produk' => ProdukVarian::with([
                'produk',
                'stokFisiks' => fn($q) => $q->where('total_panjang', '>', 0),
                'detailIsiPaket.satuan',
            ])->get(),
        ]);
    }

    public function bayar(Request $request, Penjualan $penjualan)
    {
        $data = $request->validate([
            'tanggal_bayar' => 'required|date',
            'metode_bayar' => 'required|string',
            'total_bayar' => 'required|numeric|min:0',
            'details' => 'array',
        ]);

        DB::transaction(function () use ($penjualan, $data) {

            $pembayaran = Pembayaran::create([
                'penjualan_id' => $penjualan->id,
                'tanggal_bayar' => $data['tanggal_bayar'],
                'total_bayar' => $data['total_bayar'],
                'jenis_pembayaran' => $data['metode_bayar'],
            ]);

            foreach ($data['details'] ?? [] as $detail) {
                $pembayaran->detail_pembayarans()->create($detail);
            }

            $penjualan->update([
                'status_penjualan' => 'PAID',
                'total_akhir' => $data['total_bayar'],
            ]);
        });

        return redirect()->route('ecer.penjualan.print', $penjualan);
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Pembayaran extends Model
{
    protected $fillable = [
        'penjualan_id',
        'nomer_pembayaran',
        'tanggal_bayar',
        'total_bayar',
        'jenis_pembayaran',
        'keterangan',
        'is_dp',
    ];

    public function penjualan()
    {
        return $this->belongsTo(Penjualan::class, 'penjualan_id');
    }

    public function detail_pembayarans()
    {
        return $this->hasMany(PembayaranDetails::class, 'pembayaran_id');
    }
}
